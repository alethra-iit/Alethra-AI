cmake_minimum_required(VERSION 3.22.1)
set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
project(alethra_whisper C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
# Target CPU: ARMv8.2 + dotprod + fp16 (hampir semua SoC modern)
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -march=armv8.2-a+dotprod+fp16")
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -march=armv8.2-a+dotprod+fp16")
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)

if(NOT EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/whisper.cpp/CMakeLists.txt)
  message(FATAL_ERROR "whisper.cpp belum ada. Jalankan: bash scripts/fetch-native.sh")
endif()
add_subdirectory(whisper.cpp whisper_build)

add_library(alethra_whisper SHARED whisper_jni.cpp)
target_link_libraries(alethra_whisper PRIVATE whisper android log)
target_link_options(alethra_whisper PRIVATE -Wl,--exclude-libs,ALL -Wl,-z,max-page-size=16384)
