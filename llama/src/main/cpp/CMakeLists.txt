cmake_minimum_required(VERSION 3.22.1)
# Selalu Release (bahkan untuk APK debug) supaya inferensi tidak sangat lambat.
set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
project(alethra_llama C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH "armv8.2-a+dotprod+fp16" CACHE STRING "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)

if(NOT EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/llama.cpp/CMakeLists.txt)
  message(FATAL_ERROR "llama.cpp belum ada. Jalankan: bash scripts/fetch-native.sh")
endif()
add_subdirectory(llama.cpp llama_build)

add_library(alethra_llama SHARED llama_jni.cpp)
target_link_libraries(alethra_llama PRIVATE llama android log)
target_link_options(alethra_llama PRIVATE -Wl,--exclude-libs,ALL -Wl,-z,max-page-size=16384)
